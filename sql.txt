-- Users table
CREATE TABLE Users (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(255),
    firstname VARCHAR(50),
    lastname VARCHAR(50),
    salary FLOAT,
    age INTEGER,
    registerday DATE,
    signintime DATETIME
);

-- Register (parameterized — prevents SQL injection)

VALUES (?, ?, ?, ?, ?, ?, ?, NULL);

-- Sign in (parameterized — prevents SQL injection)
SELECT * FROM Users WHERE username = ? AND password = ?;

-- Update sign-in time
UPDATE Users SET signintime = ? WHERE username = ?;

-- Requirement 3: Search by first name and/or last name
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE firstname LIKE ? AND lastname LIKE ?;

-- Requirement 4: Search by user ID
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE username = ?;

-- Requirement 5: Search by salary range
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE salary BETWEEN ? AND ?;

-- Requirement 6: Search by age range
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE age BETWEEN ? AND ?;

-- Requirement 7: Find users registered after a specific user
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday > (
    SELECT registerday
    FROM Users
    WHERE username = ?
);

-- Requirement 8: Find users who have never signed in
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE signintime IS NULL;

-- Requirement 9: Find users registered on the same day as a specific user
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday = (
    SELECT registerday
    FROM Users
    WHERE username = ?
);

-- Requirement 10: Find users registered today
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday = CURDATE();
